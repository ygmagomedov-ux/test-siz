const EXAM_COUNT = 10;
const PASS_PERCENT = 80;
const NEXT_DELAY = 300;

let order = [], pos = 0, locked = false, answers = [];
let timerInt = null, startTs = 0, userFio = '', userReason = '';

function showScreen(id) {
  document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
  document.getElementById(id).classList.add('active');
  window.scrollTo(0, 0);
}
function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}
function fmtTime(sec) {
  return String(Math.floor(sec / 60)).padStart(2, '0') + ':' + String(sec % 60).padStart(2, '0');
}
function loadJournal() {
  try { return JSON.parse(localStorage.getItem('examJournal') || '[]'); } catch (e) { return []; }
}
function saveJournal(j) { localStorage.setItem('examJournal', JSON.stringify(j)); }

function startTest() {
  userFio = document.getElementById('inp-fio').value.trim();
  userReason = document.getElementById('inp-reason').value;
  if (!userFio) { alert('Введите ФИО'); return; }
  order = shuffle(QUESTIONS.map((_, i) => i)).slice(0, Math.min(EXAM_COUNT, QUESTIONS.length));
  pos = 0; locked = false; answers = [];
  startTs = Date.now();
  clearInterval(timerInt);
  timerInt = setInterval(() => {
    document.getElementById('test-timer').textContent = fmtTime(Math.floor((Date.now() - startTs) / 1000));
  }, 1000);
  document.getElementById('test-timer').textContent = '00:00';
  document.getElementById('test-info').textContent = userFio + ' | ' + userReason;
  showScreen('screen-test');
  renderQuestion();
}

function renderQuestion() {
  const q = QUESTIONS[order[pos]];
  locked = false;
  document.getElementById('test-progress').textContent = 'Вопрос ' + (pos + 1) + '/' + order.length;
  const pf = document.getElementById('progress-fill');
  if (pf) pf.style.width = Math.round(pos / order.length * 100) + '%';
  document.getElementById('question-text').textContent = q.q;
  const list = document.getElementById('options-list');
  list.innerHTML = '';
  q.o.forEach((text, i) => {
    const btn = document.createElement('button');
    btn.className = 'option-btn';
    btn.textContent = (i + 1) + '. ' + text;
    btn.addEventListener('click', () => optionClick(i + 1, btn));
    list.appendChild(btn);
  });
}

function optionClick(num, btn) {
  if (locked) return;
  locked = true;
  answers.push({ qIndex: order[pos], chosen: num });
  btn.classList.add('selected');
  setTimeout(() => {
    try {
      pos++;
      if (pos >= order.length) finishTest();
      else renderQuestion();
    } catch (e) { alert('Ошибка: ' + e.message); }
  }, NEXT_DELAY);
}

function finishTest() {
  clearInterval(timerInt);
  const timeSec = Math.floor((Date.now() - startTs) / 1000);
  let correct = 0, badCount = 0;
  answers.forEach(a => {
    const q = QUESTIONS[a.qIndex];
    if (q && a.chosen === q.c) correct++; else badCount++;
  });
  const total = order.length || 1;
  const percent = Math.round(correct / total * 100);
  const passed = percent >= PASS_PERCENT;
  try {
    const j = loadJournal();
    j.push({
      date: new Date().toLocaleString('ru-RU'), fio: userFio, reason: userReason, mode: 'Тест',
      score: correct + ' из ' + total, percent: percent + '%',
      result: passed ? 'ЗАЧЁТ' : 'НЕ ЗАЧЁТ', time: fmtTime(timeSec)
    });
    saveJournal(j);
  } catch (e) {}
  const titleEl = document.getElementById('result-title');
  if (titleEl) titleEl.textContent = passed ? '✅ Тест сдан' : '❌ Тест не сдан';
  const box = document.getElementById('result-score');
  if (box) {
    box.className = 'score-box ' + (passed ? 'pass' : 'fail');
    box.innerHTML = 'Правильных ответов: ' + correct + ' из ' + total +
      ' (' + percent + '%)<br>Время: ' + fmtTime(timeSec) + '<br>Порог: ' + PASS_PERCENT + '%';
  }
  showScreen('screen-result');
  try {
    const rev = document.getElementById('review-list');
    if (rev) {
      rev.innerHTML = '';
      answers.forEach((a, idx) => {
        const q = QUESTIONS[a.qIndex];
        if (!q) return;
        const ok = a.chosen === q.c;
        const div = document.createElement('div');
        div.className = 'review-item ' + (ok ? 'ok' : 'bad');
        div.innerHTML =
          '<div class="rq">' + (idx + 1) + '. ' + (ok ? '✅' : '❌') + ' ' + q.q + '</div>' +
          (ok ? '' : '<div class="ra bad-text">Ваш ответ: ' + a.chosen + '. ' + q.o[a.chosen - 1] + '</div>') +
          '<div class="ra ok-text">Правильный ответ: ' + q.c + '. ' + q.o[q.c - 1] + '</div>' +
          '<div class="rt"> ' + q.topic + '</div>';
        rev.appendChild(div);
      });
      rev.classList.add('hidden');
    }
    const tbtn = document.getElementById('btn-toggle-review');
    if (tbtn) {
      tbtn.textContent = ' Показать разбор ошибок (' + badCount + ')';
      tbtn.classList.remove('hidden');
    }
  } catch (e) {}
}

function renderJournal() {
  const j = loadJournal();
  const list = document.getElementById('journal-list');
  list.innerHTML = '';
  if (j.length === 0) { list.innerHTML = '<p>Журнал пуст</p>'; return; }
  j.slice().reverse().forEach(rec => {
    const div = document.createElement('div');
    div.className = 'journal-item';
    div.innerHTML = '<b>' + rec.fio + '</b> — ' + rec.mode + '<br>' + rec.date + ' | ' + rec.reason +
      '<br>Результат: ' + rec.score + ' (' + rec.percent + ') | ' + rec.result + ' | Время: ' + rec.time;
    list.appendChild(div);
  });
}
function exportJournal() {
  const j = loadJournal();
  if (j.length === 0) { alert('Журнал пуст'); return; }
  let csv = '\uFEFFДата;ФИО;Причина;Режим;Результат;Процент;Оценка;Время\n';
  j.forEach(r => {
    csv += [r.date, r.fio, r.reason, r.mode, r.score, r.percent, r.result, r.time]
      .map(v => '"' + String(v).replace(/"/g, '""') + '"').join(';') + '\n';
  });
  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8' });
  const a = document.createElement('a');
  a.download = 'journal_result.csv';
  a.href = URL.createObjectURL(blob);
  a.click();
}
